// Smallest Prime Factor 
vector<int> sieveSPF (int n) {
  vector<int> sieve(n+1, 0);
  for(int i=2; i<=n; i++) {
    if(sieve[i] != 0) {
      continue;
    }
    for(int j=i; j<=n; j+=i) {
      if(sieve[j] == 0) {
        sieve[j] = i; 
      }
    }
  }
  return sieve;
}

// Factorize with SPF
vector<int> factorize(int n, vector<int> &sieve) {
  vector<int> factors;
  while(sieve[n]) {
    factors.push_back(sieve[n]);
    n /= sieve[n];
  }
  if(n>1) {
    factors.push_back(n);
  }
  return factors;
}

vector<int> sumOfDivisors (int n) {
  vector<int> sum(n+1, 0);
  for(int i=1; i<=n; i++) {
    for(int j=i; j<=n; j+=i) {
      sum[j] += i;
    }
  }
  return sum;
}

// Update with Prime Factor -> leads to updating with big
vector<int> biggestPrimeDivisor (int n) {
  vector<int> big(n+1, 1);
  for(int i=1; i<=n; i++) {
    if(big[i] == 1) {
      for(int j=i; j<=n; j+=i) {
        big[j] = i;
      }
    }
  }
  return big;
}
