vector<int> isPrime(int n) {
  vector<int> isPrime = new vector<int>(n+1, true);
  isPrime[0] = isPrime[1] = 0;
  for(int i = 2; i <= n; i++) {
    if(isPrime[i]) {
      for(int j=2*i; j<=n; j+=i) {
        isPrime[j] = 0;
      }
    }
  }
  return isPrime;
}
