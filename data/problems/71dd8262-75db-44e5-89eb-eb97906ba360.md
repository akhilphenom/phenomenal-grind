```cpp
class Solution {
public:
  int cleanupDataset(string dataset, int x, int y) {
    int ans = 0;
    int n = dataset.size();
    vector<int>f(26, 0);
    
    for(auto c: dataset) {
      f[c-'a']++;
    }
    
    if(x <= y) {
      int evens = 0, odds = 0;
      for(auto num: f) {
        evens += num/2;
      }
      odds = n/2 - evens;
      return x*evens + y*odds;
    } else {
      int mx = *max_element(f.begin(), f.end());
      if(mx > n/2) {
        int excessSamePairs = mx - n/2;
        int differentPairs = (n/2) - excessSamePairs;
        return excessSamePairs*x + differentPairs*y;
      }
      return (n/2)*y;
    }
    
  }
};
```
