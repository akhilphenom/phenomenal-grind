```cpp
class Solution {
public:
  int countInaccurateResults(vector<int> processOrder, vector<int> executionOrder) {
    int i=0, j=0;
    int count = 0;
    int n = processOrder.size();
    while(j<n) {
      if(processOrder[i] != processOrder[j]) {
        count++;
        j++;
      } else {
        i++;
        j++;
      }
    }
    return count;
  }
};
```
