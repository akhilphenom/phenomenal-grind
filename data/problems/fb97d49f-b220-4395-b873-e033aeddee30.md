```cpp
#include <bits/stdc++.h>

class Solution {
public:
  int makeAllElementsDistinct(int n, vector<int> identifier) {
    int idx = 0;
    unordered_map<int,int>mp;
    for(auto it: identifier) {
      mp[it]++;
    }
    for(int i=0; i<n; i++) {
      int num = identifier[i];
      if(mp[num]>1) {
        idx = i;
      }
      mp[num]--;
    }
    return idx+1;
  }
};
```
