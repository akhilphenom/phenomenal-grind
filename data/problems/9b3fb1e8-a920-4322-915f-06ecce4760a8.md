// Order
int findMinimumTimeRequired(string requests, int minGap) {
    string s;
    unordered_map<char,int>mp;
    for(int i=0; i<requests.size(); i++) {
      char ch = requests[i];
      if(mp.find(ch) == mp.end()) {
        mp[ch] = i;
        s += ch;
      } else {
        int gap = i - mp[ch] - 1;
        while(gap < minGap) {
          s += '_';
          gap--;
        }
        mp[ch] = s.size();
      }
    }
    return s.size();
}


// No Order

int findMinimumTimeRequired(string requests, int minGap) {
    unordered_map<char,int>mp;
    int mxFreq = 0;
    int countOfCharWithMxFreq = 0;
    for(auto ch: requests) {
      mp[ch]++;
      mxFreq = max(mp[ch], mxFreq);
    }

    for(auto [_, freq]: mp) {
      if(freq == mxFreq) {
        countOfCharWithMxFreq++;
      }
    }

    int time = (mxFreq - 1)*(minGap+1) + countOfCharWithMxFreq;
    return max(time, requests.size())
}
