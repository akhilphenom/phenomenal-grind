#include <bits/stdc++.h>

#define MAX 5000000

typedef unsigned long long ull;
using namespace std;

vector<int> preCompute () {
    vector<int> sieve(MAX+1);
    for(int i=0; i<=MAX; i++) {
        sieve[i] = i;
    }
    
    for(int p=2; p<=MAX; p++) {
        if(sieve[p] == p) {
            sieve[p] = p-1; // phi(p) = p-1 if p is prime
            for(int k=2*p; k<=MAX; k+=p) { //multiples are 2p,3p,...p^(k-1)p for 1..p^k
                sieve[k] = (sieve[k]/p) * (p-1);
            }
        }
    }
    return sieve;
}

vector<ull> getPrefix(vector<int> &sieve) {
    vector<ull>prefix(MAX+1, 0);
    for(int i=1; i<=MAX; i++) {
        prefix[i] = prefix[i-1] + (ull)sieve[i]*sieve[i];
    }
    return prefix;
}

int main() {
    int t, a, b;
    ull sum;
    cin>>t;
    vector<int> sieve = preCompute();
    vector<ull> prefix = getPrefix(sieve);
    for(int i=1; i<=t; i++) {
        cin>>a>>b;
        sum = prefix[b] - prefix[a-1];
        cout<<"Case "+to_string(i)+": "+to_string(sum)<<endl;
    }
    
    return 0;
}
