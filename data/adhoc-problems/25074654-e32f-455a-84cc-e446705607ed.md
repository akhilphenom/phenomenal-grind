#include <bits/stdc++.h>
#define MAX 100000

typedef unsigned long long ull;
using namespace std;

vector<int> preprocess() {
    vector<int> sieve(MAX+1);
    
    for(int i=0; i<=MAX; i++) {
        sieve[i] = i;
    }
    
    for(int p=2; p<=MAX; p++) {
        if(sieve[p] == p) {
            sieve[p] = p-1;
            for(int j = 2*p; j<=MAX; j += p) {
                sieve[j] = (sieve[j]/p)*(p-1);
            }
        }
    }
    
    return sieve;
}

int main()
{
    vector<int> sieve = preprocess(1000000);
    return 0;
}
