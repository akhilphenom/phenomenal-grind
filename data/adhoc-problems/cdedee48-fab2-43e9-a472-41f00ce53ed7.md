phi(a.b) = phi(a).phi(b)
        phi(p^k) = p^k - p^(k-1)
        1337 prime factorization = [7^1, 191^1]

Euler's Totient Application: a^n === a^(n mod phi(m)) mod m;
        if a = 2147483647 (too large, I got inf already). Thinking of 2^31-1, mostly 3 TC will pass :)
        Don't just return (int)pow(a, exponent)%m, use binary exponentiation

class Solution {
public:
    const int m = 1337;

    int calculatePhi(vector<pair<int,int>> &factors) {
        int product = 1;
        for(auto [p, k]: factors) {
            product *= pow(p,k) - pow(p, k-1);
        }
        return product;
    }

    vector<pair<int,int>> primeFactorization (int n) {
        vector<pair<int,int>> factors;
        for(int i=2; i*i<=n; i++) {
            if(n%i) {
                continue;
            }
            pair<int, int> p = {i, 0};
            while(n%i == 0) {
                p.second++;
                n/=i;
            }
            factors.push_back(p);
        }
        if(n>1) {
            factors.push_back({n,1});
        }
        return factors;
    }

    int constructExponent(vector<int> &b, int phi) {
        int exp = 0;
        for(auto d: b) {
            exp = (exp*10 + d)%phi;
        }
        if(exp == 0) {
            exp = phi;
        }
        return exp;
    }

    int modularExponentiation(int base, int exponent, int mod) {
        int result = 1;
        base %= mod;
        while(exponent) {
            if(exponent%2) {
                result = (result*base)%mod;
            }
            base = (base*base)%mod;
            exponent/=2;
        }
        return result;
    }

    int superPow(int a, vector<int>& b) {
        vector<pair<int,int>> factors = primeFactorization(m);
        int phi = calculatePhi(factors);
        int exponent = constructExponent(b, phi);
        return modularExponentiation(a, exponent, m);
    }
};
