#include <bits/stdc++.h>
#define MAX 100000

typedef unsigned long long ull;
using namespace std;

vector<int> getCoprimes (int n) {
    vector<int> coprimes;
    for(int i=1; i<n; i++) {
        if(gcd(i,n)==1) {
            coprimes.push_back(i);
        }
    }
    return coprimes;
}

vector<int> getList(vector<int> &coprimes, int n) {
    vector<int>result;
    ull product = 1;

    for(auto c: coprimes) {
        product = (product*c)%n;
    }

    if(product%n == 1) {
        result = coprimes;
    } else {
        for(auto c: coprimes) {
            if(c != n-1) {
                result.push_back(c);
            }
        }
    }
    return result;
}

int main()
{
    int n;
    cin>>n;
    vector<int> coprimes = getCoprimes(n);
    vector<int> result = getList(coprimes, n);
    cout<<result.size()<<endl;
    for(auto num: result) {
        cout<<num<<" ";
    }
    return 0;
}
